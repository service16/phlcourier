function trackPackage(trackingNumber) { db.ref("shipments/" + trackingNumber).once("value") .then((snapshot) => { const data = snapshot.val(); if (!data) { alert("Tracking number not found. Please check and try again."); return; } // Display package details on your tracking UI document.getElementById("status-display").innerText = data.currentStatus; document.getElementById("destination-display").innerText = data.destination; // Render package history timeline const historyList = document.getElementById("history-timeline"); historyList.innerHTML = ""; data.history.forEach((h) => { historyList.innerHTML += `

${h.status}

${h.location} - ${h.timestamp}

`; }); }); }
