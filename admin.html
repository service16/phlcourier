// Function to update package history and email the client instantly function updatePackageHistory(trackingNumber, newStatus, newLocation) { const dbRef = db.ref("shipments/" + trackingNumber); dbRef.once("value").then((snapshot) => { const shipment = snapshot.val(); if (!shipment) { alert("Tracking number not found!"); return; } // Create the new history entry const newEntry = { status: newStatus, location: newLocation, timestamp: new Date().toLocaleString() }; // Append to existing history array const updatedHistory = shipment.history ? [...shipment.history, newEntry] : [newEntry]; // Update Firebase Database dbRef.update({ currentStatus: newStatus, history: updatedHistory }).then(() => { // Send instant email notification via EmailJS return emailjs.send("service_q2z18lj", "jrxtuv9", { to_name: shipment.recipient || "Valued Client", to_email: shipment.clientEmail, message_title: `Shipment Status Update: ${newStatus}`, message_body: `Your package (${trackingNumber}) status has been updated to: ${newStatus}. Current Location: ${newLocation}. Log in to your tracking portal to view the full timeline.` }); }) .then(() => { alert("Package history successfully updated and notification email sent to client!"); }) .catch((err) => { console.error("Update error:", err); alert("Database updated, but email notification failed."); }); }); }
